async function uploadVideo(event) { const file = event.target.files[0]; if (!file) return; const { data: userData } = await supabaseClient.auth.getUser(); const user = userData.user; if (!user) { alert("Vous devez vous connecter avant de publier."); return; } const caption = prompt("Écrivez une description pour la vidéo :") || ""; const extension = file.name.split(".").pop(); const fileName = `${Date.now()}.${extension}`; const storagePath = `${user.id}/${fileName}`; const { error: uploadError } = await supabaseClient.storage .from("videos") .upload(storagePath, file); if (uploadError) { alert("Erreur envoi vidéo : " + uploadError.message); return; } const { data: publicUrlData } = supabaseClient.storage .from("videos") .getPublicUrl(storagePath); const videoUrl = publicUrlData.publicUrl; const { error: insertError } = await supabaseClient .from("videos") .insert({ user_id: user.id, caption: caption, video_url: videoUrl, storage_path: storagePath, status: "published" }); if (insertError) { alert("Erreur publication : " + insertError.message); return; } alert("Vidéo publiée sur VUVU ✅"); location.reload(); }
